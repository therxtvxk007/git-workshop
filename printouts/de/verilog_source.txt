@@ ITEM (A) AND gate
@@ FILE and.v
// Gate level
module and_gl (input A, B, output Y);
  and (Y, A, B);
endmodule
// Data flow
module and_df (input A, B, output Y);
  assign Y = A & B;
endmodule
// Behavioural
module and_bh (input A, B, output reg Y);
  always @(*) begin
    if (A == 1 && B == 1)
      Y = 1;
    else
      Y = 0;
  end
endmodule
@@ FILE and_tb.v
// test bench
module and_tb;
  reg A, B;
  wire Y_gl;
  wire Y_df;
  wire Y_bh;
  and_gl testa (A, B, Y_gl);
  and_df testb (A, B, Y_df);
  and_bh testc (A, B, Y_bh);
  initial begin
    A = 0; B = 0;
    #1 B = 1;
    #1 A = 1; B = 0;
    #1 B = 1;
  end
  initial begin
    $monitor("%t | A = %b | B = %b | Y_gl = %b | Y_df = %b | Y_bh = %b", $time, A,
         B, Y_gl, Y_df, Y_bh);
    $dumpfile("dump_and.vcd");
    $dumpvars();
  end
endmodule

// Data flow
module mux_dataflow (Z, D, S);
  input [3:0] D;
  input [1:0] S;
  output Z;
  assign Z = (~S[1] & ~S[0] & D[0]) | (~S[1] & S[0] & D[1]) | (S[1] & ~S[0] & D[2]) |
      (S[1] & S[0] & D[3]);

// Behavioural
module mux_behavioural (Z, D, S);
  input [3:0] D;
  input [1:0] S;
  output reg Z;
  always @(*) begin

      0: begin
        if (S[0] == 0)
          Z = D[0];
        else
          Z = D[1];
      end
      1: begin
        if (S[0] == 0)
          Z = D[2];
        else
          Z = D[3];
      end
    endcase
  end
endmodule

@@ EXPERIMENT 9 | Flip-Flops
@@ ITEM (A) SR latch
@@ FILE srlatch.v
module srlatch (input S, R, output reg Q, Qbar);
  always @(S or R) begin
    if (S == 0 && R == 0) begin
      Q = Q;
      Qbar = Qbar;
    end
    else if (S == 1 && R == 0) begin
      Q = 1;
      Qbar = 0;
    end
    else if (S == 0 && R == 1) begin
      Q = 0;
      Qbar = 1;
    end
    else begin
      Q = 1'bx;
      Qbar = 1'bx;
    end
  end
endmodule
@@ FILE srlatch_tb.v
// test bench

@@ ITEM (C) J-K flip-flop
@@ FILE jkff.v
module jkff (input J, K, CLK, output reg Q);
  always @(posedge CLK) begin
    case ({J, K})
       2'b00: begin Q = Q; end
       2'b01: begin Q = 0; end
       2'b10: begin Q = 1; end
       2'b11: begin Q = ~Q; end
    endcase
  end
endmodule

@@ EXPERIMENT 10 | Asynchronous Counters And Shift Register
@@ ITEM (A) Up counter
@@ FILE up_counter.v
// Up counter
module up_counter_tff_behavioural (input CLK, RESET, output [1:0] Q);
  wire [1:0] T;
  assign T[0] = 1;
  assign T[1] = Q[0];
  T_ff_behav T0 (.T(T[0]), .CLK(CLK), .RESET(RESET), .Q(Q[0]));
  T_ff_behav T1 (.T(T[1]), .CLK(CLK), .RESET(RESET), .Q(Q[1]));
endmodule
// T flip-flop
module T_ff_behav (input T, CLK, RESET, output reg Q);
  always @(posedge CLK or posedge RESET) begin
    if (RESET)
      Q <= 0;
    else if (T)
      Q <= ~Q;
    else
      Q <= Q;
  end
endmodule
@@ FILE up_counter_tb.v
// test bench
module up_counter_tff_behavioural_tb;
  reg CLK, RESET;
  wire [1:0] Q;
  up_counter_tff_behavioural test (.CLK(CLK), .RESET(RESET), .Q(Q));
  always #5 CLK = ~CLK;
  initial begin
    $monitor("Time = %t | RESET = %b | Count = %d | Binary = %b",
         $time, RESET, Q, Q);
    CLK = 0;
    RESET = 1;
    #10 RESET = 0;
    #100;
    RESET = 1;
    #10 RESET = 0;
    #50;
    $finish;
  end
endmodule

@@ ITEM (C) SISO shift register
@@ FILE siso.v
// D flip-flop
module d_ff (input CLK, RESET, D, output reg Q);
  always @(posedge CLK or posedge RESET) begin
    if (RESET)
       Q <= 0;
    else
       Q <= D;
  end
endmodule
// SISO shift register
module siso_shift_register (input CLK, RESET, SERIAL_IN, output SERIAL_OUT);
  wire Q0, Q1, Q2, Q3;
  d_ff d0 (.CLK(CLK), .RESET(RESET), .D(SERIAL_IN), .Q(Q0));
  d_ff d1 (.CLK(CLK), .RESET(RESET), .D(Q0), .Q(Q1));
  d_ff d2 (.CLK(CLK), .RESET(RESET), .D(Q1), .Q(Q2));
  d_ff d3 (.CLK(CLK), .RESET(RESET), .D(Q2), .Q(Q3));
  assign SERIAL_OUT = Q3;
endmodule
@@ FILE siso_tb.v
// test bench
module tb_siso_shift_register();

  wire SERIAL_OUT;
  siso_shift_register test (
       .CLK(CLK), .RESET(RESET),
       .SERIAL_IN(SERIAL_IN), .SERIAL_OUT(SERIAL_OUT)
       );
  initial begin
    CLK = 0;
    forever #5 CLK = ~CLK;
  end
  initial begin
    RESET = 1; SERIAL_IN = 0;
    #10 RESET = 0;
    SERIAL_IN = 1; #10;
    SERIAL_IN = 0; #10;
    SERIAL_IN = 1; #10;
    SERIAL_IN = 1; #10;
    SERIAL_IN = 0; #40;
    $finish;
  end
  initial begin
    $monitor("Time = %t | SERIAL_IN = %b | SERIAL_OUT = %b",
          $time, SERIAL_IN, SERIAL_OUT);
  end
endmodule
